function placeSquare(location){ if(!selectedShip){ return; } if(currentSquares.includes(location)){ return; } // First square if(currentSquares.length === 0){ currentSquares.push(location); drawBoard(); updateText(); return; } // Second square decides direction if(currentSquares.length === 1){ let first=currentSquares[0]; let row1=Math.floor(first/10); let col1=first%10; let row2=Math.floor(location/10); let col2=location%10; if(row1===row2){ direction="horizontal"; } else if(col1===col2){ direction="vertical"; } else{ document.getElementById("message").textContent = "❌ Ships must be straight!"; return; } currentSquares.push(location); if(currentSquares.length === selectedShip.size){ finishShip(); return; } drawBoard(); updateText(); return; } // Third and later squares let last=currentSquares[currentSquares.length-1]; let lastRow=Math.floor(last/10); let lastCol=last%10; let newRow=Math.floor(location/10); let newCol=location%10; if(direction==="horizontal"){ if(newRow!==lastRow || location!==last+1){ document.getElementById("message").textContent = "❌ Continue straight!"; return; } } if(direction==="vertical"){ if(newCol!==lastCol || location!==last+10){ document.getElementById("message").textContent = "❌ Continue straight!"; return; } } currentSquares.push(location); if(currentSquares.length === selectedShip.size){ finishShip(); return; } drawBoard(); updateText(); } function finishShip(){ let completedShip={ name:selectedShip.name, positions:[...currentSquares] }; shipsPlaced.push(completedShip); removeButton(selectedShip.name); document.getElementById("message").textContent = "🚢 "+completedShip.name+" Complete!"; selectedShip=null; currentSquares=[]; direction=null; checkReady(); drawBoard(); }
